# Production Operational Order Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Transform the existing production order into an auditable operational workspace covering material allocation, finished output, process execution, labor, stock movements, costs, control map, and results map.

**Architecture:** Extend the current production models without replacing `ProductionOrder` or `MaterialConsumption`. Put cross-module mutations in `production/services.py`, reuse `StockMovement` and `ProductionCostCapture`, expose thin DRF actions, and add dedicated read-only map views to the existing Bootstrap UI.

**Tech Stack:** Python 3.13, Django 5.2.16, Django REST Framework 3.17.1, PostgreSQL, Bootstrap 5, pytest-django.

## Global Constraints

- Preserve the existing single-instance architecture and Django model permissions.
- Production may receive finished goods into quarantine but may not release quality.
- Every composite stock/cost action must use `transaction.atomic()` and `select_for_update()`.
- Material reservation, issue, and finished receipt must be idempotent.
- Existing production orders and material consumptions must remain valid after migration.
- Update migrations, APIs, UI menus/actions, tests, and production documentation together.
- Do not modify or delete user-owned untracked files.

@classmethod
def issue_reserved_stock(cls, *, consumption, user=None):
    quantity = consumption.actual_quantity
    if quantity <= 0:
        raise ValidationError({'actual_quantity': 'Informe o consumo real.'})
    with transaction.atomic():
        balance = cls._get_balance_for_update(
            consumption.material,
            consumption.stock_lot,
            consumption.warehouse,
            consumption.location,
            StockQualityStatus.APPROVED,
        )
        if consumption.issue_movement_id:
            return consumption.issue_movement
        if balance.reserved_quantity < quantity:
            raise ValidationError({'reserved_quantity': 'A reserva não cobre o consumo real.'})
        cls._validate_issue(balance, quantity, allow_reserved_consumption=True)
        balance.quantity = (balance.quantity - quantity).quantize(QUANTITY_SCALE)
        balance.reserved_quantity = (
            balance.reserved_quantity - quantity
        ).quantize(QUANTITY_SCALE)
        balance.full_clean()
        balance.save(update_fields=['quantity', 'reserved_quantity', 'updated_at'])
        return cls.objects.create(
            movement_type=cls.MovementType.ISSUE,
            product=consumption.material,
            lot=consumption.stock_lot,
            quantity=quantity,
            unit=consumption.unit,
            quality_status=StockQualityStatus.APPROVED,
            from_warehouse=consumption.warehouse,
            from_location=consumption.location,
            source_production_order=consumption.order,
            source_material_consumption=consumption,
            document_reference=consumption.order.order_number,
            reason='Baixa de matéria-prima para produção',
            created_by=user,
        )

@classmethod
def receive_production_output(cls, *, output, user=None):
    if output.produced_quantity <= 0:
        raise ValidationError({'produced_quantity': 'Informe a quantidade produzida.'})
    cls._validate_address(output.warehouse, output.location)
    with transaction.atomic():
        if output.stock_movement_id:
            return output.stock_movement
        lot, _created = StockLot.objects.get_or_create(
            product=output.product,
            lot_number=output.lot_number,
            sublot_number=output.sublot_number,
            defaults={
                'quality_status': StockQualityStatus.QUARANTINE,
                'source_production_order': output.order,
                'manufacturing_date': output.manufacturing_date,
                'expiry_date': output.expiry_date,

from decimal import Decimal

from django.core.exceptions import ValidationError
from django.db import transaction
from django.utils import timezone


class ProductionOrderOperations:
    def __init__(self, order, user):
        self.order = order
        self.user = user

    def _locked_order(self):
        from production.models import ProductionOrder

        return ProductionOrder.objects.select_for_update().get(pk=self.order.pk)

    @transaction.atomic
    def reserve_materials(self):
        from inventory.models import StockMovement, StockQualityStatus

        order = self._locked_order()
        if order.status not in {order.Status.APPROVED, order.Status.RELEASED}:
            raise ValidationError({'status': 'A ordem deve estar aprovada ou liberada.'})
        movements = []
        for item in order.material_consumptions.select_related(
            'material', 'stock_lot', 'warehouse', 'location', 'unit', 'reservation_movement'
        ):
            if item.reservation_movement_id:
                movements.append(item.reservation_movement)
                continue
            if not all((item.stock_lot_id, item.warehouse_id, item.location_id)):
                raise ValidationError({'materials': f'Alocação incompleta para {item.material}.'})
            movement = StockMovement.reserve_stock(
                product=item.material,
                lot=item.stock_lot,
                warehouse=item.warehouse,
                location=item.location,
                quantity=item.planned_quantity,
                unit=item.unit,
                reason=f'Reserva para {order.order_number}',
                quality_status=StockQualityStatus.APPROVED,
            )
            movement.source_production_order = order
            movement.source_material_consumption = item
            movement.created_by = self.user
            movement.save(
                update_fields=[
                    'source_production_order',
                    'source_material_consumption',
                    'created_by',
                    'updated_at',
                ]
            )
            item.reserved_quantity = item.planned_quantity
            item.reservation_movement = movement
            item.save(
                update_fields=['reserved_quantity', 'reservation_movement', 'updated_at']
            )
            movements.append(movement)
        self._audit('production.materials.reserved', {'movements': [m.pk for m in movements]})
        return movements

    @transaction.atomic
    def issue_materials(self):
        from inventory.models import StockMovement

        order = self._locked_order()
        if order.status != order.Status.IN_PROGRESS:
            raise ValidationError({'status': 'A ordem deve estar em execução.'})
        movements = []
        for item in order.material_consumptions.select_related(
            'issue_movement', 'loss_movement', 'return_movement'
        ):
            reconciled = item.actual_quantity + item.loss_quantity + item.returned_quantity
            if reconciled != item.reserved_quantity:
                raise ValidationError({
                    'materials': (
                        f'O consumo, a perda e a devolução de {item.material} '
                        'devem reconciliar a quantidade reservada.'
                    )
                })
            changed_fields = []
            if item.actual_quantity > 0 and not item.issue_movement_id:
                movement = StockMovement.issue_reserved_stock(
                    consumption=item, user=self.user
                )
                item.issued_quantity = movement.quantity
                item.issue_movement = movement
                changed_fields.extend(['issued_quantity', 'issue_movement'])
            if item.loss_quantity > 0 and not item.loss_movement_id:
                item.loss_movement = StockMovement.record_reserved_loss(
                    consumption=item, user=self.user
                )
                changed_fields.append('loss_movement')
            if item.returned_quantity > 0 and not item.return_movement_id:
                item.return_movement = StockMovement.release_reserved_stock(
                    consumption=item, user=self.user
                )
                changed_fields.append('return_movement')
            if changed_fields:
                item.save(update_fields=[*changed_fields, 'updated_at'])
            movements.extend(
                movement
                for movement in (
                    item.issue_movement,
                    item.loss_movement,
                    item.return_movement,
                )
                if movement is not None
            )
        self._audit('production.materials.issued', {'movements': [m.pk for m in movements]})
        return movements

    @transaction.atomic
    def receive_outputs(self):
        from inventory.models import StockLotGenealogy, StockMovement

        order = self._locked_order()
        if order.status != order.Status.COMPLETED:
            raise ValidationError({'status': 'A ordem deve estar concluída.'})
        outputs = []
        input_lots = list(
            order.material_consumptions.exclude(issue_movement=None).values_list(
                'stock_lot_id', flat=True
            )
        )
        for output in order.outputs.select_related('stock_movement', 'stock_lot'):
            if not output.stock_movement_id:
                movement = StockMovement.receive_production_output(output=output, user=self.user)
                output.stock_lot = movement.lot
                output.stock_movement = movement
                output.status = output.Status.RECEIVED
                output.received_by = self.user
                output.received_at = timezone.now()
                output.save(
                    update_fields=[
                        'stock_lot',
                        'stock_movement',
                        'status',
                        'received_by',
                        'received_at',
                        'updated_at',
                    ]
                )
            for input_lot_id in input_lots:
                StockLotGenealogy.objects.get_or_create(
                    input_lot_id=input_lot_id,
                    output_lot=output.stock_lot,
                    production_order=order,
                    relation_type=StockLotGenealogy.RelationType.CONSUMED_IN_PRODUCTION,
                    defaults={
                        'quantity': next(
                            item.actual_quantity
                            for item in order.material_consumptions.all()
                            if item.stock_lot_id == input_lot_id
                        ),
                        'unit': next(
                            item.unit
                            for item in order.material_consumptions.all()
                            if item.stock_lot_id == input_lot_id
                        ),
                        'document_reference': order.order_number,
                    },
                )
            outputs.append(output)
        self._audit('production.outputs.received', {'outputs': [o.pk for o in outputs]})
        return outputs

    @transaction.atomic
    def calculate_cost(self, *, cost_center, period_start, period_end):
        from costing.models import ProductionCostCapture, StandardCost

        order = self._locked_order()
        planned_cost = Decimal('0.0000')
        material_cost = Decimal('0.0000')
        loss_cost = Decimal('0.0000')
        for item in order.material_consumptions.select_related('material'):
            standard = StandardCost.objects.filter(
                product=item.material,
                cost_center=cost_center,
                status=StandardCost.Status.APPROVED,
                effective_from__lte=period_end,
            ).order_by('-effective_from', '-version').first()
            if standard is None or standard.standard_quantity <= 0:
                raise ValidationError({
                    'cost_center': f'Custo padrão aprovado ausente para {item.material}.'
                })
            unit_cost = standard.total_standard_cost / standard.standard_quantity
            planned_cost += item.planned_quantity * unit_cost
            material_cost += item.actual_quantity * unit_cost
            loss_cost += item.loss_quantity * unit_cost
        labor_cost = sum(
            (
                entry.duration_minutes / Decimal('60.00')
            ) * entry.hourly_cost
            for entry in order.labor_entries.all()
        )
        machine_cost = sum(
            (
                operation.actual_minutes / Decimal('60.00')
            ) * operation.machine_hourly_cost
            for operation in order.operation_executions.all()
        )
        capture, _created = ProductionCostCapture.objects.get_or_create(
            production_order=order,
            cost_center=cost_center,
            period_start=period_start,
            period_end=period_end,
        )
        capture.planned_cost = planned_cost
        capture.actual_material_cost = material_cost
        capture.actual_loss_cost = loss_cost
        capture.actual_labor_cost = labor_cost
        capture.actual_machine_cost = machine_cost
        capture.full_clean()
        capture.save(update_fields=[
            'planned_cost',
            'actual_material_cost',
            'actual_loss_cost',
            'actual_labor_cost',
            'actual_machine_cost',
            'updated_at',
        ])
        capture.calculate_actuals()
        self._audit('production.cost.calculated', {'capture': capture.pk})
        return capture

    def _audit(self, action, safe_context):
        from governance.models import GovernanceAuditLog

        GovernanceAuditLog.record(
            log_type=GovernanceAuditLog.LogType.FUNCTIONAL,
            severity=GovernanceAuditLog.Severity.INFO,
            module='production',
            action=action,
            target_model='ProductionOrder',
            target_record_id=self.order.pk,
            user=self.user,
            message=f'Ação operacional executada na ordem {self.order.order_number}.',
            safe_context=safe_context,
        )
```

class ProductionOrderMapMixin(LoginRequiredMixin, TemplateView):
    template_name = 'app/production_order_map.html'
    permission_required = 'production.view_productionorder'
    map_kind = ''

    def dispatch(self, request, *args, **kwargs):
        if not request.user.has_perm(self.permission_required):
            raise PermissionDenied
        self.order = get_object_or_404(
            ProductionOrder.objects.select_related('product', 'formula', 'route', 'responsible'),
            pk=kwargs['pk'],
        )
        return super().dispatch(request, *args, **kwargs)

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context.update(
            order=self.order,
            map_kind=self.map_kind,
            materials=self.order.material_consumptions.select_related(
                'material', 'stock_lot', 'warehouse', 'location', 'unit'
            ),
            outputs=self.order.outputs.select_related('stock_lot', 'stock_movement'),
            operations=self.order.operation_executions.select_related('route_step', 'recorded_by'),
            labor_entries=self.order.labor_entries.select_related('user', 'operation_execution'),
            movements=self.order.stock_movements.select_related('product', 'lot'),
            cost_captures=self.order.cost_captures.select_related('cost_center'),
        )
        return context


class ProductionControlMapView(ProductionOrderMapMixin):
    map_kind = 'control'


class ProductionResultsMapView(ProductionOrderMapMixin):
    map_kind = 'results'

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        planned = self.order.planned_quantity
        actual = self.order.actual_yield_quantity or Decimal('0.0000')
        context['summary'] = {
            'yield_percent': (
                (actual / planned) * Decimal('100.0000')
                if planned
                else Decimal('0.0000')
            ).quantize(Decimal('0.0001')),
            'loss_quantity': self.order.real_loss_quantity,
            'rework_quantity': self.order.rework_quantity,
        }
        return context
```
